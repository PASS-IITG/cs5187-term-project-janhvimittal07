-- test/sql/aggregate/aggregates/test_quantile_disc_double.test
SELECT quantile_disc(x, 0.9) AS dec_q, quantile_disc(x, 0.9::DOUBLE) AS dbl_q FROM range(1, 11) t(x);
SELECT n,
       (SELECT quantile_disc(x, 0.9) FROM range(1, n + 1) t(x)) AS dec_q,
       (SELECT quantile_disc(x, 0.9::DOUBLE) FROM range(1, n + 1) t(x)) AS dbl_q
FROM range(8, 12) r(n) ORDER BY n;
SELECT quantile_disc(x, -0.9) AS dec_q, quantile_disc(x, -0.9::DOUBLE) AS dbl_q FROM range(1, 11) t(x);
SELECT count(*)
FROM (
    SELECT n, quantile_disc(x, {q}) AS dec_q, quantile_disc(x, {q}::DOUBLE) AS dbl_q
    FROM range(1, 201) r(n), LATERAL (SELECT x FROM range(1, n + 1) t(x)) s
    GROUP BY n)
WHERE dec_q IS DISTINCT FROM dbl_q
   OR dec_q IS DISTINCT FROM greatest(1, ceil(n * {q}))::BIGINT;
SELECT count(*)
FROM (
    SELECT n,
           (SELECT quantile_disc(x, {q}) FROM range(1, n + 1) t(x)) AS dec_q,
           (SELECT quantile_disc(x, {q}::DOUBLE) FROM range(1, n + 1) t(x)) AS dbl_q
    FROM range(1, 201) r(n))

   OR dec_q IS DISTINCT FROM greatest(1, ceil(n * {q}))::BIGINT;
